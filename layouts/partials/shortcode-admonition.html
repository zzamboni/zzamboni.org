{{- /* Keep legacy shortcode calls, but let Blowfish's blockquote hook render them. */ -}} {{- $body := strings.Trim .Inner "\r\n" -}} {{- $title := strings.TrimSpace (.Title | default "") -}} {{- $alert := printf "[!%s]" (strings.ToUpper .Name) -}} {{- with $title -}} {{- $alert = printf "%s %s" $alert . -}} {{- end -}} {{- $markdown := printf "> %s\n%s" $alert (replaceRE "(?m)^" "> " $body) -}} {{- with .Icon -}} {{- $markdown = printf "%s\n{icon=%q}" $markdown . -}} {{- end -}} {{- .Page.RenderString (dict "display" "block") $markdown -}}
