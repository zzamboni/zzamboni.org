{{- $resume := . -}} {{- $options := $resume.meta.themeOptions | default dict -}} {{- $labels := $options.sectionLabels | default dict -}} {{- $order := $options.sections | default (slice "work" "education" "projects" "certificates" "awards" "publications" "skills" "languages" "references") -}} {{- $sections := slice -}} {{- range $order -}} {{- $key := . -}} {{- $name := index $labels $key | default (cond (hasPrefix $key "projects:") (strings.TrimPrefix "projects:" $key) ($key | humanize)) -}} {{- $items := slice -}} {{- if hasPrefix $key "projects:" -}} {{- $group := strings.TrimPrefix "projects:" $key -}} {{- range $resume.projects -}} {{- if eq .type $group -}}{{- $items = $items | append . -}}{{- end -}} {{- end -}} {{- else if eq $key "projects" -}} {{- $items = $resume.projects | default (slice) -}} {{- else -}} {{- $items = index $resume $key | default (slice) -}} {{- end -}} {{- with $items -}} {{- $sections = $sections | append (dict "id" (replace $key ":" "-" | anchorize) "key" $key "label" $name "entries" .) -}} {{- end -}} {{- end -}} {{- return $sections -}}
