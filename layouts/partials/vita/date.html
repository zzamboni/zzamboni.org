{{- /* JSON Resume allows dates with year-only precision in this CV. Avoid parsing YYYY as a full timestamp. */ -}} {{- $value := .value | default "" | string -}} {{- $format := .format | default "Jan 2006" -}} {{- if $value -}} {{- if findRE `^[0-9]{4}$` $value -}} {{- $value -}} {{- else -}} {{- (time $value).Format $format -}} {{- end -}} {{- end -}}
